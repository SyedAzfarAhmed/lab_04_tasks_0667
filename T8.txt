#include <stdio.h>
int main()
{
    int pizzas, crust;
    printf("Enter the number of pizzas: ");
    scanf("%d", &pizzas);
    printf("Enter crust type: ");
    scanf("%d", &crust);
    if (pizzas==3 && crust==3)
    {
        printf("You get free garlic bread!");
    }
    else
    {
        printf("Thank you for your order!");
    }
}
